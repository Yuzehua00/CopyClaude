  159:         cmd = EventSubscribeCommand.model_validate(params)

  172:     # 从 events.jsonl 向 writer 回放匹配 topic 的历史事件，返回已回放条数
  173:     async def _replay_events(
  174:         self,
  175:         run_id: str,
  176:         writer: asyncio.StreamWriter,
  177:         topics: list[str],
  178:     ) -> int:
  179:         path = events_file(run_id) # 根据run_id获取事件文件路径event_json。
  180:         if not path.exists(): # 路径不存在就去C盘精确匹配。
  181:             for candidate in Path("~/.kama/sessions").expanduser().glob(
  182:                 f"*/runs/{run_id}/events.jsonl"
  183:             ):
  184:                 path = candidate
  185:                 break
  186:         if not path.exists(): # 匹配未成功则返回。
  187:             return 0
  188: 
  189:         count = 0
  190:         for line in path.read_text().splitlines(): # 按行读取文件
  191:             if not line:
  192:                 continue
  193:             try:
  194:                 event = json.loads(line)
  195:             except json.JSONDecodeError:
  196:                 continue
  197:             event_type: str = event.get("type", "")
  198:             if not any(fnmatch.fnmatch(event_type, p) for p in topics):
  199:                 continue
  200:             envelope = EventPushEnvelope(event=event)
  201:             writer.write(envelope.model_dump_json().encode() + b"\n")
  202:             count += 1
  203: 
  204:         if count:
  205:             await writer.drain()
  206:         return count
